  // the rules run, so everything else in the text is still checked.
  function check(text, allow) {
    if (allow) text = text.replace(new RegExp(allow.source, "gi"), " ");
    const hits = [];
    for (const r of RULES) {
      const m = text.match(r.re);
      if (m) hits.push({ rule: r.name, word: m[0].trim(), why: r.why });
    }
    const n = text.match(NAME_RE);
    if (n) hits.push({ rule: "reference name", word: n[2], why: "names never go in the prompt" });
    return hits;
  }

  return { RULES, NAMES, check };

/* ONE OF ONE — keys and chord loops. Minor-heavy, hip hop canon only. */
"use strict";

const OneTheory = (() => {
  const SHARP = ["C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"];
  const FLAT = ["C", "Db", "D", "Eb", "E", "F", "Gb", "G", "Ab", "A", "Bb", "B"];
  const PC = { C: 0, "C#": 1, Db: 1, D: 2, "D#": 3, Eb: 3, E: 4, F: 5, "F#": 6, Gb: 6, G: 7, "G#": 8, Ab: 8, A: 9, "A#": 10, Bb: 10, B: 11 };
  const SHARP_KEYS = { minor: new Set(["E", "B", "F#", "C#", "G#", "D#", "A#"]), major: new Set(["G", "D", "A", "E", "B", "F#"]) };
  const MINOR_KEYS = SHARP.slice();
  const MAJOR_KEYS = FLAT.slice();

  // [semitones above tonic, suffix]. Short loops; the riff does the talking.
  const LOOPS = [
    { tonality: "minor", roman: "i – ♭VI", chords: [[0, "m"], [8, ""]] },
    { tonality: "minor", roman: "i – ♭II", chords: [[0, "m"], [1, ""]] },
    { tonality: "minor", roman: "i (pedal)", chords: [[0, "m"]] },
    { tonality: "minor", roman: "i – ♭VI – V", chords: [[0, "m"], [8, ""], [7, ""]] },
    { tonality: "minor", roman: "i – ♭VII – ♭VI – V", chords: [[0, "m"], [10, ""], [8, ""], [7, ""]] },
    { tonality: "minor", roman: "i – iv", chords: [[0, "m"], [5, "m"]] },
    { tonality: "minor", roman: "i7 – IV7", chords: [[0, "m7"], [5, "7"]] },
    { tonality: "minor", roman: "i9 – ♭VImaj7", chords: [[0, "m9"], [8, "maj7"]] },
    { tonality: "minor", roman: "i – v – ♭VI", chords: [[0, "m"], [7, "m"], [8, ""]] },
    { tonality: "major", roman: "IVmaj7 – iii7 – vi9", chords: [[5, "maj7"], [4, "m7"], [9, "m9"]] },
    { tonality: "major", roman: "I – ♭VII", chords: [[0, ""], [10, ""]] },
  ];

  const name = (pc, tonic, tonality) => (SHARP_KEYS[tonality].has(tonic) ? SHARP : FLAT)[((pc % 12) + 12) % 12];

  function harmony(r, pick, avoidKeys = [], majorShare = 0.15) {
    // Minor most of the time; a genre pack or lane can raise the major share.
    // One roll decides the tonality (filtering per loop would call r() eleven times and drown the share).
    const wantMajor = r() < majorShare;
    const pool = LOOPS.filter((l) => l.tonality === (wantMajor ? "major" : "minor"));
    const loop = pick(pool.length ? pool : LOOPS, r);
    const keys = loop.tonality === "minor" ? MINOR_KEYS : MAJOR_KEYS;
    let tonic = pick(keys, r);
    for (let i = 0; i < 8 && avoidKeys.includes(`${tonic} ${loop.tonality}`); i++) tonic = pick(keys, r);
    const chords = loop.chords.map(([o, q]) => name(PC[tonic] + o, tonic, loop.tonality) + q);
    return { key: `${tonic} ${loop.tonality}`, roman: loop.roman, chords };
  }

  return { harmony };
})();

// The short drum word for {KICK} {HATS} {SNARE}: the words up to and including
// the drum noun. "a syncopated double kick before the snare" -> "syncopated double kick".
OnePersonality.drumWord = function drumWord(source) {
  const m = source.replace(/^an? /, "").match(/^(.*?\b(kick|hats?|snare|clap|shaker|snap)s?)\b/i);
  return m ? m[1] : source.replace(/^an? /, "").split(",")[0];
};

// Render a layout with a values map. Null values take `fallbacks`; a
// placeholder with nothing left is dropped and the comma list is tidied.
OnePersonality.render = function render(layout, values) {
  const F = OnePersonality.fallbacks;
  const resolve = (key, depth = 0) => {
    const v = values[key];
    if (v) return v;
    const fb = F[key];
    if (!fb || depth > 3) return "";
    const m = fb.match(/^\{(\w+)\}$/);
    return m ? resolve(m[1], depth + 1) : fb;
  };
  return layout.sections
    .map((s) => s.replace(/\{(\w+)\}/g, (_, k) => resolve(k)).replace(/,\s*(?=,|\])/g, "").replace(/\|\s*,/, "|").replace(/\bidentical, identical\b/g, "identical"))
    .join("\n\n");
};

/* ONE OF ONE — one engine. Builds every sound from grammar slots, resolves
 * clashes, and lays the result out in Suno's own field order.
 *
 * Style field order (Suno weights the front and reads it as tags):
 *   head: anchor, BPM, key, SYNC, TWITCH, SLUR, SWITCH or LOOP, LOW LEAD,
 *         lead handle, mix tag, instrumental            (~220 chars, never cut)
 *   prose: lead (every slot), bass, drums, counter       (trimmed from the end)
 *
 * Modes: switch (one axis, its words repeated in the head, the cue and every
 * later section), loop (identical [Loop] cues, nothing arranged), straight.
 * "Every N bars" phrases are capped at one per roll and zero in loop mode,
 * because they teach Suno to change something every four bars.
 *
 * A genre pack (src/one/genres.js) overrides whole sections of the base
 * grammar and may supply its own STRUCTURE(p, r, h) for the lyrics field. */

  // Vocal packs unlock the voice words the instrumental guard bans, and drop the vocal exclusions.
  // chant, choir, crowd, gospel and anthem stay banned even here: they summon group voices.
  const VOICE_WORDS = /\b(vocal\w*|vocalist|sing\w*|sung|hum|humming|hummed|lyric\w*|ad-?lib\w*|falsetto|harmonies|rap|rapper|rapped|rapping|voices?|doubles|spoken|whisper\w*|belt\w*|auto-?tune\w*|melismatic|hook)\b/i;
  const VOCAL_EXCLUDE_DROP = /^(vocals|singing|rap|humming|background vocals|chanting)$/i;

  function mulberry32(a) {
    return () => {
      a |= 0;
      a = (a + 0x6d2b79f5) | 0;
      let t = Math.imul(a ^ (a >>> 15), 1 | a);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }
  const pick = (arr, r) => arr[Math.floor(r() * arr.length)];
  // Pick from the entries that pass every rule; fall back to the whole list.
  const pickWhere = (arr, r, ok) => {
    const pool = arr.filter(ok);
    return pick(pool.length ? pool : arr, r);
  };
  const int = (lo, hi, r) => lo + Math.floor(r() * (hi - lo + 1));
  const weighted = (entries, r) => {
    const total = entries.reduce((n, [, w]) => n + w, 0);
    let x = r() * total;
    for (const [v, w] of entries) if ((x -= w) < 0) return v;
    return entries[entries.length - 1][0];
  };
  // "a drunken detuned organ ...|the drunken organ" -> [full, handle]
  const split = (entry) => entry.split("|");
  const bare = (s) => s.replace(/^an? /, "");
  // Phrases that teach Suno to change something every four bars.
  const EVERY = /every (\d+\w*|other|third) (bar|pass|4|8)|\bbar \d+( and bar \d+)?|every \d+ bars|partway/i;

  const grammarFor = (genreId) => {
    const pack = Genres[genreId] || Genres.hiphop;
    return { ...Base, ...pack.grammar, id: pack.id, label: pack.label };
  };

  // Crossbreed: the lead, melody and structure from one genre, the drums, bass
  // and groove from another, picked fresh each roll. Nothing in a pack can
  // express it, so the engine stitches two grammars together.
  function crossbreed(r) {
    const ids = Object.keys(Genres).filter((id) => id !== "cross");
    const a = pick(ids, r);
    const b = pick(ids.filter((id) => id !== a), r);
    const A = grammarFor(a), B = grammarFor(b);
    const allow = [A.ALLOW, B.ALLOW].filter(Boolean);
    return {
      ...A,
      id: "cross", label: Genres.cross.label,
      cross: { lead: A.label, drums: B.label, drumsTag: B.DRUMS_TAG },
      DRUMS: B.DRUMS, BASS: B.BASS, GROOVE: B.GROOVE, COLOR: B.COLOR, TWITCH: B.TWITCH,
      TWITCH_LABEL: B.TWITCH_LABEL, TWITCH_FROM_KICK: B.TWITCH_FROM_KICK,
      RHYTHM: { ...A.RHYTHM, sync: B.RHYTHM.sync, twitch: B.RHYTHM.twitch, slur: B.RHYTHM.slur },
      SLUR_SHARE: B.SLUR_SHARE,
      MIX_TAG: A.MIX_TAG.split(",")[0],
      EXCLUDE: [...new Set([...A.EXCLUDE, ...B.EXCLUDE])],
      ALLOW: allow.length ? new RegExp(allow.map((x) => x.source).join("|"), "i") : undefined,
      GUEST_SHARE: 0,
    };
  }

  // One "every N bars" phrase per roll, none in loop mode.
  function makeAllow(mode) {
    let left = mode === "loop" ? 0 : 1;
    const fn = (phrase) => {
      if (EVERY.test(phrase)) left--;
    };
    fn.peek = () => left > 0;
    return fn;
  }
  const pickBudget = (arr, r, allow, ok = () => true) => {
    const chosen = pickWhere(arr, r, (x) => ok(x) && (!EVERY.test(x) || allow.peek()));
    allow(chosen);
    return chosen;
  };

  // Guest leads: a spin-off genre borrows the experimental hip hop palette
  // (found sounds, mutant synths, fuzz guitar, drunk organ, flipped wind, low
  // brass) for the lead some of the time. The counter then stays in-genre so
  // the record keeps its identity.
  const GUEST_FAMILIES = [["found", 3], ["synth", 3], ["guitar", 2], ["organ", 2], ["wind", 1], ["brass", 1]];

  function buildLead(r, G, lane, allow) {
    const L = G.LEAD;
    const guest = (G.GUEST_SHARE || 0) > 0 && r() < G.GUEST_SHARE;
    const family = guest ? weighted(GUEST_FAMILIES, r) : weighted(Object.entries(lane.families), r);
    const F = guest ? { ...Base.FAMILIES[family], frame: G.GUEST_FRAME || Base.FAMILIES[family].frame } : G.FAMILIES[family];
    const [source, handle] = split(pick(F.sources, r));
    const lowFirst = r() < G.LOW_SHARE;
    const register = pickWhere(L.register, r, (x) => (lowFirst ? !/^mid register/.test(x) : true));
    const low = !/^mid register|low-mid, stepping/.test(register);
    const motif = pickBudget(L.motif, r, allow);
    const long = /five|six|seven|run/.test(motif);
    const short = /two-note|one repeated|stubborn root/.test(motif);
    const articulation = pickBudget(L.articulation, r, allow, (a) =>
      !(long && /one held|one anticipated|a push note|two stabs|three-note pickup/.test(a)) &&
      !(short && /three-note pickup/.test(a)));
    const legato = /legato|long tones|held note/.test(articulation);
    const placement = pick(L.placement, r);
    const dry = /dry|narrow|mono/.test(placement);
    const character = pickWhere(F.character, r, (c) =>
      !(dry && /chorus|detuned/.test(c)) &&
      !(legato && /gated/.test(c)) &&
      !(low && /band-passed/.test(c)) &&
      !(/detuned/.test(source) && /detuned|chorus|ring-mod/.test(c)));
    const motion = pickBudget(L.motion, r, allow, (m) =>
      !(dry && /delay feedback|detune widens/.test(m)) &&
      !(legato && /shorter/.test(m)) &&
      !(/bitcrush|grittier/.test(m) && /low-passed/.test(character)));
    return {
      family, guest, handle, frame: F.frame, source, character, register, articulation, placement, motion, motif,
      human: pick(L.human, r),
      low, dry,
      triplet: /triplet|three-against/.test(articulation),
      is808: /^an 808/.test(source),
    };
  }

  function buildBass(r, G, lead, lane, allow) {
    const B = G.BASS;
    const fits = (s) => !(lead.low && /fingered|saw basses|tuba|sousaphone/.test(s)) && !(/tuba|sousaphone/.test(s) && /tuba|sousaphone/.test(lead.source));
    const laneBass = lane.bass || [];
    const laneFits = laneBass.filter(fits);
    // A lane's own basses (the South Side tuba) come up about half the time.
    const pool = laneFits.length && r() < 0.5 ? laneFits : B.source;
    const source = lead.is808
      ? pick(["a clean sine sub", "a sub that only sustains on the root"], r)
      : pickBudget(pool, r, allow, fits);
    const patterns = lane.bassPattern || B.pattern;
    const pattern = pickBudget(patterns, r, allow, (p) =>
      !(lead.low && /motif|moving when/.test(p)) &&
      !(lead.triplet && /triplet/.test(p)) &&
      !(lead.is808 && !lane.bassPattern && !/one note|silent|three notes|root/.test(p)));
    return { source, tone: pick(B.tone, r), pattern, human: pick(B.human, r) };
  }

  function buildCounter(r, G, lead, lane, allow, mode) {
    const forced = lane.counterFamily && lead.family !== lane.counterFamily && r() < 0.5;
    // A guest lead gets an in-genre counter, so the genre's own sound is always in the record.
    const others = forced ? [[lane.counterFamily, 1]] : Object.entries(lane.families).filter(([f]) => lead.guest || f !== lead.family);
    const family = others.length ? weighted(others, r) : pick(Object.keys(G.FAMILIES).filter((f) => f !== lead.family), r);
    const F = G.FAMILIES[family];
    const [source, handle] = split(pickWhere(F.sources, r, (x) => !/808/.test(x)));
    const role = pickBudget(G.COUNTER_ROLE, r, allow, (x) => !(mode !== "switch" && /after the switch/.test(x)));
    return { family, source, handle, character: pick(F.character, r), role };
  }

  function buildDrums(r, G, lead, allow, mode, lane = {}) {
    const D = { ...G.DRUMS, ...(lane.drums || {}) };
    const hats = pickBudget(D.hats, r, allow, (h) => !(lead.triplet && /threes/.test(h)));
    const twitch = {
      sound: pick(G.TWITCH.sound, r),
      rhythm: pickBudget(G.TWITCH.rhythm, r, allow, (x) => !(lead.triplet && /threes|triple/.test(x))),
      where: pickBudget(G.TWITCH.where, r, allow, (w) => !(mode !== "switch" && /after the switch/.test(w))),
    };
    // One stumbling element only: when the hats roll in threes the slur goes elsewhere.
    const slur = pickWhere(G.GROOVE.triplet, r, (t) => !(/threes/.test(hats) && /hats/.test(t)));
    return {
      color: pickBudget(G.COLOR, r, allow),
      kick: pickBudget(D.kick, r, allow),
      snare: pickBudget(D.snare, r, allow),
      hats,
      twitch,
      sync: pickBudget(G.GROOVE.sync, r, allow),
      feel: pick(G.GROOVE.human, r),
      slur,
    };
  }

  function buildSwitch(r, G, lead, bass, drums, lane = {}) {
    const S = G.SWITCH;
    const axis = pickWhere(lane.switchAxes || S.axes, r, (a) =>
      !(/half-time/.test(a.tag) && /half-time/.test(drums.snare)) &&
      !(/strips/.test(a.cue) && /after the switch/.test(drums.twitch.where)));
    const sparse = /strips/.test(axis.cue);
    const land = pickWhere(S.land, r, (l) =>
      !(sparse && /drums join|everything slams/.test(l)) &&
      !(/half-time/.test(axis.tag) && /two bars later/.test(l)));
    return { ...axis, land };
  }

  function buildRhythm(r, G, mode, lead, sw, allow) {
    const R = G.RHYTHM;
    const modeTag = mode === "switch" ? sw.tag : mode === "loop" ? pick(R.loop, r) : pick(R.straight, r);
    return {
      sync: pickBudget(R.sync, r, allow),
      twitch: pickBudget(R.twitch, r, allow),
      slur: r() < G.SLUR_SHARE ? pick(R.slur, r) : null,
      mode: modeTag,
      low: lead.low ? pick(R.low, r) : null,
    };
  }

  // The first clause of the processing phrase, short enough for the head: "pitched down a whole step".
  const timbre = (character) => {
    const first = character.split(",")[0].trim();
    // A long first clause would only fit as a fragment: the handle then stands alone.
    if (first.split(/\s+/).length > 6) return "";
    const words = first.split(/\s+/).slice(0, 6);
    while (words.length > 2 && /^(the|a|an|on|of|with|and|in|to|at|by|so|like|its|for|into|over|under|every|each|until|then|all|that|this|more|too|very|no|never)$/i.test(words[words.length - 1])) words.pop();
    return words.join(" ");
  };

  function style(p, G) {
    const { lead, bass, drums, rhythm } = p;
    const voiceTag = p.vocal ? p.vocal.tag : G.VOCAL ? "vocals up front" : "instrumental";
    const head = [
      `${p.anchor}, ${p.bpm} BPM, ${p.key}`, G.VOCAL ? null : voiceTag,
      rhythm.sync, rhythm.twitch, rhythm.slur, p.mode === "loop" || (p.layout && p.layout.minimal >= 0.7) ? "minimal" : null, rhythm.mode, p.extraTag, rhythm.low,
      `${lead.handle} ${timbre(lead.character)}`.trim(), p.mixTag || G.MIX_TAG, G.VOCAL ? voiceTag : null,
    ].filter(Boolean).join(", ") + ".";
    // A source that already states its figure ("a two-note hook", "a four-chord loop") gets no second one.
    const figured = /\b(\w+-note|\w+-chord|arp|riff|loop|hook|stab|figure|melody)\b/.test(lead.source);
    const plays = figured ? lead.articulation : `playing ${lead.motif} as ${lead.articulation}`;
    // A compact pack leaves the register to the head's low tag and the slur to the head's slur tag.
    const reg = G.COMPACT && rhythm.low ? "" : `, ${lead.register}`;
    const leadLine = `lead: ${lead.source}, ${lead.character}${reg}, ${plays}, ${lead.human}, ${lead.placement}.`;
    const leadNoPlace = `lead: ${lead.source}, ${lead.character}${reg}, ${plays}, ${lead.human}.`;
    const leadShort = `lead: ${lead.source}, ${lead.character}${reg}, ${plays}.`;
    // The head already says the lead is low; the shortest lead line keeps the sound, its processing and the figure.
    const leadMin = figured ? `lead: ${lead.source}, ${lead.character}.` : `lead: ${lead.source}, ${lead.character}, playing ${lead.motif}.`;
    const bassMin = `bass: ${bass.source}.`;
    const bassLine = `bass: ${bass.source}, ${bass.tone}, ${bass.human}.`;
    const bassShort = `bass: ${bass.source}, ${bass.human}.`;
    const slur = rhythm.slur && !G.COMPACT ? `, ${drums.slur}` : "";
    const anti = p.anti ? `, ${p.anti}` : "";
    const drumLine = `drums: ${drums.kick}, ${drums.snare}, ${drums.hats}, ${drums.color}${slur}${anti}.`;
    const drumShort = `drums: ${drums.kick}, ${drums.snare}, ${drums.hats}${slur}.`;
    const noCounter = p.mode === "loop" || G.NO_COUNTER || p.noCounter;
    const counterLine = noCounter ? null : `counter: ${p.counter.source}, ${p.counter.character}, ${p.counter.role}.`;
    const counterShort = noCounter ? null : `counter: ${p.counter.source}, ${p.counter.role}.`;
    // The head and the lead always stay. Cut order: counter, bass tone,
    // percussion color, lead placement, lead human.
    const tries = [
      [head, leadLine, bassLine, drumLine, counterLine],
      [head, leadLine, bassLine, drumLine, counterShort],
      [head, leadLine, bassShort, drumLine, counterShort],
      [head, leadNoPlace, bassShort, drumLine, counterShort],
      [head, leadLine, bassLine, drumLine],
      [head, leadLine, bassShort, drumLine],
      [head, leadLine, bassShort, drumShort],
      [head, leadNoPlace, bassShort, drumShort],
      [head, leadShort, bassShort, drumShort],
      [head, leadShort, bassShort, `drums: ${drums.kick}, ${drums.snare}, ${drums.hats}.`],
      [head, leadMin, bassShort, drumShort],
      [head, leadMin, bassMin, drumShort],
      [head, leadMin, bassMin, `drums: ${drums.kick}, ${drums.snare}, ${drums.hats}.`],
      [head, leadMin, bassMin, `drums: ${drums.kick}, ${drums.snare}.`],
      [head, leadShort, bassShort],
    ];
    let s;
    for (const t of tries) if ((s = t.filter(Boolean).join(" ")).length <= 1000) break;
    // Last resort: the head without the lead's timbre clause.
    if (s.length > 1000) {
      const headShort = head.replace(`${lead.handle} ${timbre(lead.character)}`.trim(), lead.handle);
      for (const t of tries) if ((s = [headShort, ...t.slice(1)].filter(Boolean).join(" ")).length <= 1000) break;
    }
    return s;
  }

  function renderLayout(p) {
    const { lead, drums, sw, bass, counter, rhythm, layout } = p;
    const vals = {
      LEAD: lead.handle, COUNTER: counter.handle, BASS: /808/.test(bass.source) ? "808" : "bass",
      KICK: bare(drums.kick), HATS: bare(drums.hats), SNARE: bare(drums.snare), SYNC: rhythm.sync || drums.sync, SLUR: rhythm.slur || null,
      FEEL: drums.feel, MOTION: lead.motion, CUE: sw.cue, AFTER: sw.after, LAND: sw.land,
      PLACE: lead.placement.split(",")[0], TWITCH: `${drums.twitch.sound} ${drums.twitch.rhythm}`, ROLE: counter.role,
      BASSPAT: bass.pattern, ANTI: p.anti || "the same loop the whole way",
    };
    if (Personality && Personality.render) {
      const lim = Personality.limits || {};
      // Keep as many whole clauses as fit the limit; never cut mid-word. The after-words stay intact:
      // the head's switch tag and the lyrics must carry the same words.
      const clip = (k, v) => {
        if (!v || !lim[k] || v.length <= lim[k] || k === "AFTER") return v;
        const parts = v.split(", ");
        let out = parts[0];
        for (const q of parts.slice(1)) {
          if (`${out}, ${q}`.length > lim[k]) break;
          out = `${out}, ${q}`;
        }
        // A first clause that does not fit is dropped whole, so the fallback speaks instead of a cut-off phrase.
        return out.length > lim[k] ? null : out;
      };
      const shaped = { ...vals, KICK: Personality.drumWord(drums.kick), HATS: Personality.drumWord(drums.hats), SNARE: Personality.drumWord(drums.snare) };
      for (const k of Object.keys(shaped)) shaped[k] = clip(k, shaped[k]);
      const out = Personality.render(layout, shaped).replace(/\[Intro \| ([^\]]*), 4 bars\]/, (m, body) => `[Intro | ${body}, ${p.introBars} bars]`);
      return `[Instrumental]\n\n${/\[End\]/.test(out) ? out : `${out}\n\n[End]`}`;
    }
    const lines = layout.sections.map((t) => t.replace(/\{(\w+)\}/g, (m, k) => (vals[k] != null ? vals[k] : m)));
    if (!/\[End\]/.test(lines[lines.length - 1])) lines.push("[End]");
    return lines.join("\n\n");
  }

  function lyrics(p, G, r) {
    const h = { bare, int, pick };
    if (p.layout) return renderLayout(p);
    if (G.STRUCTURE) {
      const custom = G.STRUCTURE(p, r, h);
      if (custom) return custom;
    }
    const { lead, drums, sw, bass, counter, rhythm } = p;
    const L = lead.handle;
    const bassWord = /808/.test(bass.source) ? "808" : "bass";
    const twitchCue = G.TWITCH_FROM_KICK
      ? `${G.TWITCH_LABEL || "kick trick"}: ${bare(drums.kick)}, ${drums.twitch.where}`
      : `${G.TWITCH_LABEL || "twitch percussion"}: ${drums.twitch.sound} ${drums.twitch.rhythm}, ${drums.twitch.where}`;
    const slurCue = rhythm.slur ? drums.slur : null;
    const counterLate = /after the switch/.test(counter.role);

    if (p.mode === "loop") {
      const loop = `${L} playing ${bare(lead.articulation)}, ${bare(drums.kick)}, ${bare(drums.hats)}, ${bassWord} ${bass.pattern}`;
      return [
        `[Intro | ${L} alone, ${lead.placement.split(",")[0]}, 2 bars]`,
        `[Loop | same two-bar loop, ${loop}, everything else identical]`,
        `[Loop | same loop, identical, ${twitchCue}${slurCue ? `, ${slurCue}` : ""}]`,
        "[Loop | same loop, identical, hats cut for two bars then back]",
        `[Loop | same loop, ${lead.motion}, drums identical]`,
        `[Loop | same loop, identical, ${drums.feel}]`,
        "[Outro | the loop runs until it stops dead on beat 1]",
        "[End]",
      ].join("\n\n");
    }

    const verse = `[Verse | full drums, ${rhythm.sync}, ${bassWord} ${bass.pattern}${slurCue ? `, ${slurCue}` : ""}, ${twitchCue}]`;
    if (p.mode === "straight") {
      const hook = `[Hook | full beat, ${L} up front, ${lead.motion}${p.noCounter ? "" : `, ${counter.handle} ${counter.role}`}]`;
      return [
        `[Intro | ${L} alone, ${lead.placement.split(",")[0]}]`,
        verse,
        hook,
        `[Verse | same drums, ${L} thinner, ${drums.feel}]`,
        hook,
        `[Outro | drums cut, ${L} alone, ends dry on beat 1]`,
        "[End]",
      ].join("\n\n");
    }

    // Switch mode: one axis, repeated in every section after the switch.
    return [
      `[Intro | ${L} alone, ${lead.placement.split(",")[0]}]`,
      verse,
      `[Hook | full beat, ${L} up front${counterLate ? "" : `, ${counter.handle} ${counter.role}`}]`,
      `[Break | drums stop, ${sw.land}]`,
      `[Beat Switch | ${sw.cue}]`,
      `[Verse | ${sw.after}, ${L} stays low, ${counterLate ? `${counter.handle} enters, ` : ""}${/after the switch/.test(drums.twitch.where) ? "twitch percussion enters" : "twitch percussion in the gaps"}]`,
      `[Hook | ${sw.after}, heavier, ${L} up front, ${lead.motion}]`,
      `[Outro | ${sw.after}, drums cut, ends dry on beat 1]`,
      "[End]",
    ].join("\n\n");
  }

  function generate({ seed = 1, genre = "hiphop", mode, avoidKeys = [] } = {}) {
    const r = mulberry32(seed >>> 0);
    const G = genre === "cross" ? crossbreed(r) : grammarFor(genre);
    const lane = weighted(G.LANES.map((l) => [l, l.weight]), r);
    // A lane may lean major (the bounce lane does) and carry its own mix tag.
    const harmony = T.harmony(r, pick, avoidKeys, lane.majorShare != null ? lane.majorShare : G.MAJOR_SHARE);
    // A lane may carry its own tempo and mode mix (the Prof moods do).
    const [lo, hi] = pick(lane.tempo || G.TEMPO, r);
    // A pack with LAYOUTS lets the lyrics layout decide the mode (minimal loop, ballad, A-then-B...).
    let layout = null;
    if (G.LAYOUTS && !mode) {
      const allowed = lane.layouts ? G.LAYOUTS.filter((l) => lane.layouts.includes(l.id)) : G.LAYOUTS;
      // A lane may re-weight the layouts it allows (the weird lane leans hook-first).
      layout = weighted(allowed.map((l) => [l, (lane.layoutWeights && lane.layoutWeights[l.id]) || l.weight || 1]), r);
    }
    const chosenMode = mode || (layout ? layout.mode : weighted(lane.modes || G.MODES, r));
    const allow = makeAllow(chosenMode);
    const lead = buildLead(r, G, lane, allow);
    const counter = buildCounter(r, G, lead, lane, allow, chosenMode);
    const bass = buildBass(r, G, lead, lane, allow);
    const drums = buildDrums(r, G, lead, allow, chosenMode, lane);
    const sw = buildSwitch(r, G, lead, bass, drums, lane);
    const rhythm = buildRhythm(r, G, chosenMode, lead, sw, allow);
    const p = {
      anchor: G.cross ? `${pick(lane.anchors, r)}, crossed with ${G.cross.drumsTag}`
        : lane.neutralAnchor && lead.family !== lane.mainFamily ? lane.neutralAnchor : pick(lane.anchors, r),
      bpm: int(lo, hi, r),
      key: `in ${harmony.key}`,
      mode: chosenMode,
      lane: lane.id,
      layout,
      extraTag: lane.tags ? pick(lane.tags, r) : layout && G.LAYOUT_TAGS && G.LAYOUT_TAGS[layout.id] ? pick(G.LAYOUT_TAGS[layout.id], r) : null,
      anti: G.ANTI ? pick(G.ANTI, r) : null,
      mixTag: lane.neutralMixTag && lead.family !== lane.mainFamily ? lane.neutralMixTag : lane.mixTag,
      noCounter: !!(G.NO_COUNTER || lane.noCounter),
      introBars: pick([2, 4, 4, 8], r),
      lead, counter, bass, drums, sw, rhythm,
    };
    if (G.VOCAL && Lyrics) {
      const song = Lyrics.write({ r, pick, lane: lane.id, key: harmony.key, bpm: p.bpm });
      p.vocal = { tag: song.style || song.persona.tag, lyrics: song.lyrics, hook: song.hook, title: song.title, persona: song.persona, theme: song.theme };
      // The song owns its structure: the head tag follows whether it wrote a beat switch.
      p.mode = /\[Beat Switch/.test(song.lyrics) ? "switch" : "straight";
      p.rhythm.mode = p.mode === "switch" ? sw.tag : pick(G.RHYTHM.straight, r);
    }
    const styleText = style(p, G);
    const lyricsText = p.vocal ? p.vocal.lyrics : lyrics(p, G, r);
    const allowRe = G.VOCAL ? new RegExp([G.ALLOW && G.ALLOW.source, VOICE_WORDS.source].filter(Boolean).join("|"), "i") : G.ALLOW;
    const warnings = Guard.check(`${styleText}\n${lyricsText}`, allowRe).map((h) => ({ ...h, message: `${h.rule}: "${h.word}" ${h.why}` }));
    const S = (lane.sliders) || G.SLIDERS[chosenMode] || G.SLIDERS.switch;

    const styleInfluence = int(S.styleInfluence[0], S.styleInfluence[1], r);
    const model = lane.model || G.MODEL || (chosenMode === "switch" && r() < 0.5 ? "v6-wild" : "v6");
    return {
      style: styleText,
      lyrics: lyricsText,
      exclude: (G.VOCAL ? G.EXCLUDE.filter((x) => !VOCAL_EXCLUDE_DROP.test(x)) : G.EXCLUDE).join(", "),
      sliders: { weirdness, styleInfluence },
      settings: { model, variety: "Off", maxMode: "On", instrumental: G.VOCAL ? "Off" : "On" },

        layout: layout ? layout.label : undefined,
        title: p.vocal ? p.vocal.title : `${pick(G.TITLE.a, r)} ${pick(G.TITLE.b, r)}`, bpm: p.bpm, key: harmony.key,
        hook: p.vocal ? p.vocal.hook : undefined, vocal: p.vocal ? p.vocal.persona.tag : undefined,
      },
      fingerprint: [G.id, chosenMode === "loop" ? "loop" : counter.source, lead.source, lead.character, lead.motif, lead.articulation, lead.human, chosenMode === "switch" ? sw.tag : chosenMode, harmony.key].join("|"),
      parts: { lead, counter, bass, drums, switch: sw, rhythm },
      warnings,
    };
  }

  const genres = () => Object.values(Genres).filter((g) => g.show !== false).map((g) => ({ id: g.id, label: g.label, blurb: g.blurb }));

(() => {
  const E = OneEngine;
  const $ = (id) => document.getElementById(id);
  const STORE = "oneofone.seen.v1";
  let pkg;

  let seen = new Set();
  try {
    seen = new Set(JSON.parse(localStorage.getItem(STORE) || "[]"));
  } catch {
    seen = new Set();
  }
  const remember = (fp) => {
    seen.add(fp);
    try {
      localStorage.setItem(STORE, JSON.stringify([...seen].slice(-5000)));
    } catch {
      /* storage blocked: the in-memory set still prevents repeats this visit */
    }
  };
  const recentKeys = [];
  const GENRE_KEY = "oneofone.genre.v1";
  let genre = "hiphop";
  try {
    const saved = localStorage.getItem(GENRE_KEY);
    if (saved && E.genres().some((g) => g.id === saved)) genre = saved;
  } catch {
    /* default genre */
  }

  function renderGenres() {
    const row = $("genres");
    row.textContent = "";
    row.hidden = E.genres().length < 2;
    if (!E.genres().some((g) => g.id === genre)) genre = E.genres()[0].id;
    for (const g of E.genres()) {
      const b = document.createElement("button");
      b.type = "button";
      b.className = "genre";
      b.setAttribute("role", "radio");
      b.setAttribute("aria-checked", String(g.id === genre));
      b.textContent = g.label;
      b.onclick = () => {
        genre = g.id;
        try {
          localStorage.setItem(GENRE_KEY, genre);
        } catch {
          /* fine */
        }
        renderGenres();
        roll();
      };
      row.appendChild(b);
    }
    $("blurb").textContent = E.genres().find((g) => g.id === genre).blurb;
  }

  // Suno collapses similar descriptions into one sound, so the page also keeps the
  // lead family and the bass type from repeating across the last three rolls.
  const recentSounds = [];
  function roll() {
    for (let i = 0; i < 60; i++) {
      pkg = E.generate({ seed: Math.floor(Math.random() * 4294967295), genre, avoidKeys: recentKeys });
      const fam = pkg.parts.lead.family, bass = pkg.parts.bass.source, d = pkg.parts.drums;
      const clash = recentSounds.some((s) => s.fam === fam || s.bass === bass || s.kick === d.kick || s.snare === d.snare || s.hats === d.hats);
      if (!seen.has(pkg.fingerprint) && (!clash || i > 40)) break;
    }
    recentSounds.push({ fam: pkg.parts.lead.family, bass: pkg.parts.bass.source, kick: pkg.parts.drums.kick, snare: pkg.parts.drums.snare, hats: pkg.parts.drums.hats });
    if (recentSounds.length > 3) recentSounds.shift();
    remember(pkg.fingerprint);
    recentKeys.push(pkg.meta.key);
    if (recentKeys.length > 4) recentKeys.shift();
    render();
    resetSequence();
  }

  const STEPS = [

    return [
      "LYRICS", pkg.lyrics, "",
      "STYLES", pkg.style, "",
      "EXCLUDE STYLES", pkg.exclude, "",
      "SLIDERS", sliderLine(), settingsLine(),

  }

  function renderBar() {
    const btn = $("next");
    const kicker = $("next-kicker");
    const label = $("next-label");
    const count = $("next-step");
    for (const sec of document.querySelectorAll(".field[data-field]")) {
      const k = sec.dataset.field;
      sec.classList.toggle("active", step < STEPS.length ? STEPS[step].key === k : k === "sliders");
      sec.classList.toggle("pasted", pasted.has(k));
    }
    if (step < STEPS.length) {
      const s = STEPS[step];
      const prev = step > 0 ? STEPS[step - 1] : null;
      btn.classList.remove("done");
      kicker.textContent = prev ? `${prev.label} is on the clipboard. Paste it into ${prev.where} in Suno, then:` : "Tap, switch to Suno, paste. Repeat for each field.";
      label.textContent = `COPY ${s.label}`;
      count.textContent = `${step + 1} / ${STEPS.length}`;
      btn.setAttribute("aria-label", `Copy ${s.label.toLowerCase()} for Suno, step ${step + 1} of ${STEPS.length}`);
    } else {
      btn.classList.add("done");
      kicker.textContent = `EXCLUDE STYLES is on the clipboard. Paste it, then set the sliders by hand:`;
      label.textContent = sliderLine();
      count.textContent = settingsLine();
      btn.setAttribute("aria-label", "All three fields copied. Set the sliders, then tap to start over.");
    }
  }

  function advance() {
    if (step >= STEPS.length) {
      step = 0;
      pasted = new Set();
      renderBar();
      return;

    const s = STEPS[step];
    copyText(pkg[s.key]);
    pasted.add(s.key);

  }

    step = 0;
    pasted = new Set();

  }

    e.preventDefault();

    $("install").hidden = false;
  });
  $("install").onclick = async () => {
    if (!installEvent) return;
    installEvent.prompt();
    await installEvent.userChoice.catch(() => {});
    installEvent = null;
    $("install").hidden = true;
  };
  window.addEventListener("appinstalled", () => {
    $("install").hidden = true;
  });

  $("roll").onclick = roll;
  renderGenres();
  roll();
})();

test("every grammar and genre entry passes the guard", () => {
  const walk = (o, path) => {
    if (typeof o === "string") assert.deepStrictEqual(Guard.check(o), [], `${path}: ${o}`);
    else if (o && typeof o === "object") for (const k of Object.keys(o)) if (k !== "EXCLUDE" && typeof o[k] !== "function") walk(o[k], `${path}.${k}`);
  };
  walk(G, "G");
  const VOICE = /\b(vocal\w*|vocalist|sing\w*|sung|hum|humming|hummed|lyric\w*|ad-?lib\w*|falsetto|harmonies|rap|rapper|rapped|rapping|voices?|doubles|spoken|whisper\w*|belt\w*|auto-?tune\w*|melismatic|hook)\b/i;
  for (const pack of Object.values(require("../src/one/genres.js"))) {
    const allow = pack.grammar.VOCAL
      ? new RegExp([pack.grammar.ALLOW && pack.grammar.ALLOW.source, VOICE.source].filter(Boolean).join("|"), "i")
      : pack.grammar.ALLOW;
    const walkAllow = (o, path) => {
      if (typeof o === "string") assert.deepStrictEqual(Guard.check(o, allow), [], `${path}: ${o}`);
      else if (o && typeof o === "object" && !(o instanceof RegExp)) for (const k of Object.keys(o)) if (k !== "EXCLUDE" && k !== "label" && typeof o[k] !== "function") walkAllow(o[k], `${path}.${k}`);
    };
    walkAllow(pack.grammar, `genres.${pack.id}`);
  }
});

test("the lead gets every slot, up front", () => {
  for (const seed of seeds) {
    const { style, lyrics, parts } = E.generate({ seed });
    const L = parts.lead;
    // A source that states its own figure ("a two-note hook") carries no second motif.
    const figured = /\b(\w+-note|\w+-chord|arp|riff|loop|hook|stab|figure|melody)\b/.test(L.source);
    for (const slot of ["source", "character", "register", "motif", "articulation"]) {
      if (slot === "motif" && figured) continue;
      assert.ok(style.includes(L[slot]), `${slot} missing: ${style}`);
    }
    // The human-feel clause is the last thing trimmed; it may only go when the field is full.
    assert.ok(style.includes(L.human) || style.length >= 960, `human missing with room to spare: ${style}`);
    // Placement always reaches Suno: in the style field or the intro tag.
    assert.ok(style.includes(L.placement) || lyrics.includes(L.placement.split(",")[0]), style);
    assert.ok(style.indexOf("lead:") < style.indexOf("bass:") && style.indexOf("bass:") < style.indexOf("drums:"), style);
    const head = style.slice(0, 260);
    for (const slot of ["sync", "twitch", "mode", "slur", "low"]) if (parts.rhythm[slot]) assert.ok(head.includes(parts.rhythm[slot]), `${slot} not in the front tags: ${style}`);
    assert.ok(head.includes(L.handle), `lead handle not in the front tags: ${style}`);
  }
});

test("the lyrics field is section tags only: a real beat switch, or one hypnotic loop", () => {

  }
  assert.ok(loops > seeds.length * 0.28 && loops < seeds.length * 0.42, `${loops} loops`);
});
